--------------------------------------
BASTA: Binary And Simple Text Archive
--------------------------------------

BNF Grammar
-----------

Basta is a superset of Pasta.  The grammar is identical to Pasta's except
that the value production gains one additional alternative: blob.

basta               : blank , containers ;
containers          : blank , container-seq | blank , section-seq ;
container-seq       : container , { blank , container } ;
section-seq         : section , { blank , section } ;
section             : "@" , label , blank , container ;

container           : map | array ;

map                 : "{" , members , "}" ;
array               : "[" , elements , "]" ;

members             : blank | member-list ;
member-list         : member , { blank , "," , blank , member } ;
member              : blank , label , blank , ":" , blank , element ;

(* A map key -- and likewise a section name (see `section`) -- is a `label`,
   not a `value`.  A label spelled like a keyword or a number is taken as that
   label here: { true: false } has key `true` (a label) mapping to the boolean
   false; { 0: 1 } has key `0`; @0 { ... } names a section `0`.  Only
   unquoted-labels qualify unquoted -- every character must be a labelchar --
   so a signed spelling like -5 is not a bare key (quote it: "-5"). *)

elements            : blank | elements-list ;
elements-list       : element , { blank , "," , blank , element } ;
element             : blank , value , blank ;

value               : container | type | void | label-ref | blob ;
type                : string | boolean | number ;
void                : "null" ;

(* Blob: binary escape from the text domain.
   A blob may appear anywhere a value may appear.  It is introduced by
   the byte 0x00, which is illegal in all other positions in a Basta
   or Pasta document -- not in labels, not in strings (including
   multiline strings), not in any token.  The sentinel is followed by
   exactly 8 bytes encoding the blob length as an unsigned big-endian
   64-bit integer, then exactly that many bytes of arbitrary binary
   data (which may itself contain 0x00 bytes freely).  The parser
   resumes normal Pasta text immediately after the last data byte.

   Because 0x00 is only dispatched when the lexer is in value-start
   position, a zero byte that appears inside an already-open string
   scan cannot be mistaken for a blob sentinel: the string scanner
   runs in a separate mode and never invokes the value dispatch. *)
blob                : NUL , uint64be , { byte } ;
NUL                 : 0x00 ;
uint64be            : byte , byte , byte , byte , byte , byte , byte , byte ;
                    (* 8 bytes, big-endian unsigned, giving blob length N.
                       The N data bytes follow immediately. *)
byte                : (* any byte value 0x00..0xFF *) ;

(* All remaining productions are identical to Pasta. *)

(* Disambiguation in VALUE position.
   A maximal run of labelchars (see unquoted-label) can match more than one
   alternative of `value`.  Precedence during parsing is:

       keyword (true, false, null, Inf, NaN)  >  number  >  label-ref

   A lexeme that spells a keyword is that keyword; otherwise a lexeme that
   spells a valid number is that number; otherwise it is a label-ref (naming
   a section for alforno link resolution).  Keywords and numbers are never
   treated as references.  Examples:  true -> boolean,  0 -> number,
   123abc -> label-ref (not a valid number, hence a label).

   A blob is orthogonal to all of the above: it is dispatched solely by the
   0x00 sentinel, which is not a labelchar and cannot begin a label or number.

   This precedence applies ONLY in value position.  A map key and a section
   name are `label`, not `value` (see member), so no keyword/number competitor
   exists there. *)
label-ref           : unquoted-label ;

label               : unquoted-label | quoted-label ;
unquoted-label      : labelchar , { labelchar } ;
quoted-label        : "\"" , { stringchar } , "\"" ;

(* unquoted-label is tokenised by MAXIMAL MUNCH: the longest run of labelchars
   from the current position.  A run that begins with a digit but is not a
   valid number is therefore a single label, never a number followed by a
   second token.  This covers runs with trailing non-numeric labelchars
   (123abc, 0x1fg, 1_000, 12.3.4) AND runs the `number` production rejects on
   their own: a leading zero (007, 08 -- see `integer`), an empty 0x/0b prefix
   (0x, 0b), or a dot with no fractional digit (1.).  Because '-' is not a
   labelchar, a '-'-led run that is not a valid number (-, -007, -abc) is an
   error, not a label; a well-formed signed number (-5, -Inf) is a number. *)

string              : simple-string | multiline-string ;
simple-string       : "\"" , { stringchar } , "\"" ;
multiline-string    : "\"\"\"" , { anychar-but-triple-quote } , "\"\"\"" ;

boolean             : "true" | "false" ;
number              : float-constant | hex-number | bin-number
                    | signed-integer , [ "." , digits ] ;
float-constant      : "Inf" | "-Inf" | "NaN" ;
hex-number          : [ "-" ] , "0" , ( "x" | "X" ) , hexdigit , { hexdigit } ;
bin-number          : [ "-" ] , "0" , ( "b" | "B" ) , bindigit , { bindigit } ;
signed-integer      : [ "-" ] , integer ;
integer             : "0" | natural , { digit } ;
digits              : digit , { digit } ;
hexdigit            : digit | "a" | "b" | "c" | "d" | "e" | "f"
                    | "A" | "B" | "C" | "D" | "E" | "F" ;
bindigit            : "0" | "1" ;

(* Character classes *)

stringchar          : labelchar | nonalphanumeric ;
labelchar           : alphabet | digit | labelsymbol ;
alphabet            : uppercase | lowercase ;
digit               : "0" | natural ;
natural             : "1" | "2" | "3" | "4" | "5" | "6" | "7" | "8" | "9" ;
labelsymbol         : "!" | "#" | "$" | "%" | "&" | "." | "_" ;
nonalphanumeric     : "'" | "(" | ")" | "*" | "+" | "," | "-" | "/" | ":"
                    | ";" | "<" | "=" | ">" | "?" | "@" | "[" | "\" | "]" | "^"
                    | "`" | "{" | "|" | "}" | "~" ;
blank               : { nonprint | comment } ;
comment             : line-comment | block-comment ;
line-comment        : ( ";" | "//" ) , { anychar-but-newline } ;
block-comment       : "/*" , { anychar-but-close-delimiter } , "*/" ;

(* Comments are blank: they may appear anywhere `blank` may.  A block comment
   may span newlines and does not nest; it ends at the first "*/".  A block
   comment left unterminated at end of input does not match block-comment and
   is therefore a parse error.  Comment delimiters have no meaning inside a
   string or multiline string -- "http://x" is a URL, not a comment -- because
   the string scanner runs in its own mode.  Outside a string, '/' is not a
   labelchar and begins no other token, so it is only ever a comment lead-in
   or an error.  Comment scanning is also never entered inside a blob: blob
   bytes are consumed by length, not scanned as text. *)
nonprint            : EOL | 0x20 | 0x09 ;
EOL                 : 0x0A | 0x0D ;
empty               : ;
